C++ — это C с кучей дополнений. std::cout << отправляет текст на экран, std::endl заканчивает строку.

#include <iostream>

int main() {
    std::cout << "Hello, world!" << std::endl;
    return 0;
}

02_ПЕРЕМЕННЫЕ02/08

Как и в C, у каждой переменной есть тип. std::string хранит текст. Через << можно выводить много значений подряд.

#include <iostream>
#include <string>

int main() {
    std::string name = "Shiro";
    int age = 18;
    std::cout << "Hi, I'm " << name << " and I'm " << age << "\n";
}
ВЫВОД

06_ФУНКЦИИ06/08

У функции есть тип результата, название и параметры. const std::string& значит «прочитать этот текст, не копируя его».

#include <iostream>
#include <string>

std::string greet(const std::string& name) {
    return "Hello, " + name + "!";
}

int main() {
    std::cout << greet("Shiro") << "\n";
}
ВЫВОД

07_ВЕКТОРЫ07/08

std::vector — список, который может расти. push_back добавляет элемент, size() — сколько их. for (... : langs) проходит по каждому элементу.

#include <iostream>
#include <string>
#include <vector>

int main() {
    std::vector<std::string> langs = {"Python", "C++", "Rust"};
    langs.push_back("C#");
    std::cout << langs.size() << "\n";

    for (const auto& lang : langs) {
        std::cout << "I like " << lang << "\n";
    }
}

08_МИНИ-ПРОЕКТ_ГЕРОЙ_ИГРЫ08/08

Класс объединяет данные и функции в одно целое. Так в играх описывают игроков, врагов и предметы.

#include <iostream>
#include <string>

class Player {
public:
    std::string name;
    int hp = 100;

    void hit(int damage) {
        hp -= damage;
        std::cout << name << " has " << hp << " HP\n";
    }
};

int main() {
    Player p;
    p.name = "Shiro";
    p.hit(30);
    p.hit(25);
}
ВЫВОД
Shiro has 70 HP
Shiro has 45 HP

01. Что выведет этот код?

std::string s = "VIBE";
std::cout << s.length() << "\n";

В слове "VIBE" 4 буквы.

02. Что выводит текст на экран?

cout << — вывод, cin >> — ввод.

03. Что выведет этот код?

std::vector<int> v = {1, 2};
v.push_back(3);
std::cout << v.size() << "\n";

push_back добавил ещё один элемент — их стало 3.

04. Что выведет этот код?

int hp = 100;
hp -= 30;
hp -= 25;
std::cout << hp << "\n";

100 - 30 = 70, потом 70 - 25 = 45.
